/* M3 – Pán-Európai Logisztika (P8 modul). Kiegészítő modul a Solaris háttérfigyelőjéhez (m3-run.mjs.txt).
   ZÉRÓ-BLÖFF ELV: csak ténylegesen lekért, friss mérésből számol. Ami nem mérhető, az null, és a lefedettséget (V) csökkenti.
   Nincs beégetett, szimulált vagy becsült érték.

   Csatornák:
     rhine_kaub     Rajna, Kaub vízállás (PEGELONLINE, hivatalos, ingyenes, kulcs nélkül)
     danube_passau  Duna, Passau vízállás (PEGELONLINE; a német Duna-szakasz, NEM a magyar)
     border_hu      magyar határátkelők: még nincs értelmezett forrás, ezért mindig null (a rendőrség Határinfó oldala emberi olvasatra készül)
   G (stationarity_G): a mért csatornák G-jének súlyozott átlaga. Vízállásnál a hivatalos hajózhatósági határokhoz (NSW/HSW) viszonyít,
      ha az API megadja; ha nem, a hivatalos MNW/MHW-állapotból durván (jelölve: basis). 1,0 = a hajózási sávban.
   V (validity_V): a mért csatornák súlyainak aránya az összes csatornáéhoz (lefedettség). S = 1 − V. atm_score = G·S·(1−V).
   Felbontás: óránként egy rögzítés (a Solaris óránként commitol), ezért a live_24h valójában ≤ 25 rekord; a felső korlát 96.
   Sűrítés: live_24h (≤96) → daily_30d (≤30, naptári nap, folyamatos összesítéssel) → monthly_12m (≤12). Fix méret, 50 KB alatt. */
import fs from "node:fs";

export const FILE = "data/metrics/logistics_matrix.json";
export const BASE = "https://www.pegelonline.wsv.de/webservices/rest-api/v2/stations";
export const CHANNELS = [
  { id: "rhine_kaub", label: "Rajna · Kaub", type: "gauge", stations: ["KAUB"] },
  { id: "danube_passau", label: "Duna · Passau (Németország)", type: "gauge", stations: ["PASSAU DONAU", "PASSAU"] },
  { id: "border_hu", label: "Magyar határátkelők (Röszke, Tompa, Záhony)", type: "none", stations: [] },
];
export const LIMITS = { live: 96, daily: 30, monthly: 12 };
const STALE_MS = 6 * 36e5;

const clamp = (x, a = 0, b = 1) => Math.max(a, Math.min(b, x));
const num = (v) => (typeof v === "number" && Number.isFinite(v) ? v : null);
const r3 = (v) => (v === null || v === undefined ? null : Math.round(v * 1000) / 1000);
const readJ = (f, d) => { try { return JSON.parse(fs.readFileSync(f, "utf8")); } catch (e) { return d; } };

/* Vízállás → G. A hivatalos NSW (legalacsonyabb hajózható) és HSW (legmagasabb hajózható) szinthez viszonyít. */
export function gaugeG({ value, nsw = null, hsw = null, stateMnwMhw = null }) {
  const v = num(value);
  if (v === null) return { g: null, basis: null };
  if (num(nsw) !== null && num(hsw) !== null && nsw > 0 && hsw > nsw) {
    if (v < nsw) return { g: r3(clamp(v / nsw, 0.05, 1)), basis: "NSW/HSW" };
    if (v > hsw) return { g: r3(clamp(1 - (v - hsw) / 100, 0.05, 1)), basis: "NSW/HSW" };
    return { g: 1, basis: "NSW/HSW" };
  }
  const st = typeof stateMnwMhw === "string" ? stateMnwMhw.toLowerCase() : "";
  if (st === "normal") return { g: 1, basis: "MNW/MHW-állapot (durva)" };
  if (st === "low" || st === "high") return { g: 0.6, basis: "MNW/MHW-állapot (durva)" };
  return { g: null, basis: null };
}

async function getJson(url, ms = 15000) {
  const ac = new AbortController(), t = setTimeout(() => ac.abort(), ms);
  try {
    const res = await fetch(url, { signal: ac.signal, headers: { "User-Agent": "solaris-yantra-m3" } });
    if (!res.ok) throw new Error("HTTP " + res.status);
    return await res.json();
  } finally { clearTimeout(t); }
}

/* Egy mérőállomás: a jelöltek sorban, az első működő számít. Visszatér: mért csatorna-adat vagy null (és a hiba oka). */
export async function readGauge(ch, nowMs, get = getJson) {
  let lastErr = "nincs állomás";
  for (const st of ch.stations) {
    try {
      const j = await get(`${BASE}/${encodeURIComponent(st)}/W.json?includeCurrentMeasurement=true&includeCharacteristicValues=true`);
      const cm = j && j.currentMeasurement, val = cm ? num(cm.value) : null, ts = cm ? Date.parse(cm.timestamp) : NaN;
      if (val === null || !Number.isFinite(ts)) { lastErr = "nincs aktuális mérés"; continue; }
      if (nowMs - ts > STALE_MS) { lastErr = "a mérés elavult (> 6 óra)"; continue; }
      const cv = {};
      for (const c of (Array.isArray(j.characteristicValues) ? j.characteristicValues : [])) if (c && c.shortname && num(c.value) !== null) cv[String(c.shortname).toUpperCase()] = c.value;
      const { g, basis } = gaugeG({ value: val, nsw: cv.NSW ?? null, hsw: cv.HSW ?? null, stateMnwMhw: cm.stateMnwMhw });
      return { ok: true, data: { station: st, cm: val, measured_at: cm.timestamp, state: cm.stateMnwMhw ?? null, nsw: cv.NSW ?? null, hsw: cv.HSW ?? null, g, basis } };
    } catch (e) { lastErr = String((e && e.message) || e).slice(0, 80); }
  }
  return { ok: false, error: lastErr };
}

/* Csatornákból a modul G/V értéke. weights: [rhine, danube, border]. */
export function combine(channels, weights = [1, 1, 1]) {
  let wAll = 0, wOk = 0, sum = 0;
  CHANNELS.forEach((c, i) => {
    const w = Number.isFinite(weights[i]) && weights[i] > 0 ? weights[i] : 0;
    wAll += w;
    const x = channels[c.id];
    if (x && num(x.g) !== null) { wOk += w; sum += w * x.g; }
  });
  const G = wOk > 0 ? sum / wOk : null, V = wAll > 0 ? wOk / wAll : 0;
  return { G: G === null ? null : r3(G), V: r3(V) };
}

/* Folyamatos összesítés (idempotens az óránkénti őrrel): egy új rekord beépítése a napi és a havi szintbe. */
function accumulate(item, entry) {
  item.n = (item.n || 0) + 1;
  item.v_sum = r3((item.v_sum || 0) + entry.V);
  if (entry.G !== null) {
    item.n_g = (item.n_g || 0) + 1;
    item.g_sum = r3((item.g_sum || 0) + entry.G);
    item.G_min = item.G_min === undefined || item.G_min === null ? entry.G : Math.min(item.G_min, entry.G);
  }
  item.G_avg = item.n_g ? r3(item.g_sum / item.n_g) : null;
  item.V_avg = r3(item.v_sum / item.n);
  return item;
}

export function fold(dp, entry) {
  const live = Array.isArray(dp.live_24h) ? dp.live_24h.slice() : [], daily = Array.isArray(dp.daily_30d) ? dp.daily_30d.slice() : [], monthly = Array.isArray(dp.monthly_12m) ? dp.monthly_12m.slice() : [];
  live.push(entry);
  const date = entry.t.slice(0, 10), month = entry.t.slice(0, 7);
  if (daily.length && daily[daily.length - 1].date === date) accumulate(daily[daily.length - 1], entry);
  else daily.push(accumulate({ date }, entry));
  if (monthly.length && monthly[monthly.length - 1].month === month) accumulate(monthly[monthly.length - 1], entry);
  else monthly.push(accumulate({ month }, entry));
  return { live_24h: live.slice(-LIMITS.live), daily_30d: daily.slice(-LIMITS.daily), monthly_12m: monthly.slice(-LIMITS.monthly) };
}

export function buildPayload({ dp, channels, errors, G, V, nowMs, weights }) {
  const S = r3(1 - V);
  const flags = [];
  const rk = channels.rhine_kaub;
  if (rk && num(rk.g) !== null && rk.g < 0.6) flags.push(rk.nsw !== null && rk.cm < rk.nsw ? "river_low_water" : "river_high_water");
  const status = G === null ? "NO_DATA" : G < 0.6 ? "ANOMALY" : "NOMINAL";
  return {
    header: { module_id: "M3_LOGISTICS", version: "1.0.0", timestamp: Math.floor(nowMs / 1000), p8_signature: "PHYSICAL_GROUND_TRUTH" },
    core_metrics: { status, stationarity_G: G, noise_S: S, validity_V: V, atm_score: G === null ? null : r3(G * S * (1 - V)) },
    dynamic_payload: {
      ...dp,
      channels: Object.fromEntries(CHANNELS.map((c) => [c.id, { label: c.label, ...(channels[c.id] || { cm: null, g: null }), ...(errors[c.id] ? { error: errors[c.id] } : {}) }])),
      coverage: `${CHANNELS.filter((c) => channels[c.id] && num(channels[c.id].g) !== null).length}/${CHANNELS.length}`,
      flags,
      notes: ["A határátkelők csatornája nincs értelmezve (nincs hivatalos nyílt forrás), ezért null.", "A Duna-csatorna a német szakasz (Passau), nem a magyar."],
    },
    weights_vector: weights,
  };
}

export async function main(get = getJson) {
  const nowMs = Date.now(), iso = new Date(nowMs).toISOString(), hour = iso.slice(0, 13);
  const prev = readJ(FILE, null), dp = (prev && prev.dynamic_payload) || {};
  const live = Array.isArray(dp.live_24h) ? dp.live_24h : [];
  if (live.length && live[live.length - 1].hour === hour && !process.env.M3_FORCE) { console.log("M3: ebben az órában már rögzítve, kihagyom."); return prev; }
  const wf = readJ("data/metrics/weights_M3.json", null), wl = Array.isArray(wf) ? wf : wf && wf.weights_vector;
  const weights = Array.isArray(wl) && wl.length >= 3 && wl.slice(0, 3).every((x) => Number.isFinite(x) && x >= 0) ? wl.slice(0, 3) : [1, 1, 1];
  const channels = {}, errors = {};
  await Promise.all(CHANNELS.filter((c) => c.type === "gauge").map(async (c) => {
    const res = await readGauge(c, nowMs, get);
    if (res.ok) channels[c.id] = res.data; else { channels[c.id] = { cm: null, g: null }; errors[c.id] = res.error; }
  }));
  channels.border_hu = { cm: null, g: null };
  const { G, V } = combine(channels, weights);
  const entry = { t: iso, hour, G, V, ch: Object.fromEntries(CHANNELS.map((c) => [c.id, channels[c.id] && num(channels[c.id].g) !== null ? { cm: channels[c.id].cm ?? null, g: channels[c.id].g } : null])) };
  const out = buildPayload({ dp: fold(dp, entry), channels, errors, G, V, nowMs, weights });
  fs.mkdirSync("data/metrics", { recursive: true });
  fs.writeFileSync(FILE, JSON.stringify(out, null, 1));
  console.log(`M3: G ${G === null ? "nincs mérés" : G} · V ${V} (lefedettség ${out.dynamic_payload.coverage}) · ${out.core_metrics.status}${Object.keys(errors).length ? " · hiba: " + Object.entries(errors).map(([k, v]) => k + ": " + v).join("; ") : ""}`);
  return out;
}
if (process.argv[1] && process.argv[1].endsWith("m3-run.mjs")) await main();
